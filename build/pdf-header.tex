  listing engine=listings,

  colback=CodeBackground,
  colframe=CodeBorder,
  coltext=CodeForeground,
  arc=2.2mm,
  outer arc=2.2mm,
  boxrule=0.5pt,
  left=2.8mm,
  right=2.8mm,
  top=2.2mm,
  bottom=2.2mm,
  before skip=8pt,
  after skip=8pt,
  listing options={
    basicstyle=\ttfamily\small\color{CodeForeground},
    breaklines=true,
    breakatwhitespace=false,
    columns=fullflexible,
    keepspaces=true,
    showstringspaces=false,
    upquote=true
  }
}
